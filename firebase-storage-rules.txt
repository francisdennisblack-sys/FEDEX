// Recommended Firebase Storage rules for media uploads to `posts/*`.
// - Allows public reads for media URLs.
// - Allows authenticated users to write to `posts/{postId}/...` without
//   imposing a file-size limit on the client-side. Server-side size limits
//   (e.g., Cloud Storage quotas) still apply.
// - Ensures uploaded objects are images or videos by checking contentType.
// Review and tighten these rules before deploying to production.

rules_version = '2';
service firebase.storage {
  match /b/{bucket}/o {

    // Posts media: authenticated users may upload images or videos.
    match /posts/{postId}/{allPaths=**} {
      // Allow anyone to read media URLs (public-facing content)
      allow read: if true;

      // Allow writes only for authenticated users and only when the
      // uploaded object's contentType looks like an image or video.
      // Note: No explicit file-size restriction here — client may upload
      // large videos (resumable uploads handled by the client).
      allow write: if request.auth != null
                   && request.resource != null
                   && (
                     request.resource.contentType.matches('image/.*')
                     || request.resource.contentType.matches('video/.*')
                   );
    }

    // Deny writes to other paths by default; allow reads for media where appropriate.
    match /{allPaths=**} {
      allow read: if true;
      allow write: if false;
    }
  }
}
