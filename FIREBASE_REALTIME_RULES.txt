{
  "rules": {
    // Online users tracking - ANYONE can read and write (MVP testing)
    "onlineUsers": {
      "$uid": {
        ".read": true,
        ".write": true
      }
    },
    
    // Public posts - creation allowed but area tag becomes immutable
    "posts": {
      "$postId": {
        ".read": true,
        ".write": "(
            // CREATE: new post must include a textual zoneTag and not be a raw coordinate string
            (!data.exists() && newData.hasChild('zoneTag') && newData.child('zoneTag').isString() && !newData.child('zoneTag').val().matches('.*\\\d+\\\.\\\d+.*'))
            // UPDATE: only allow if the zoneTag remains unchanged (area tags are part of the post)
            || (data.exists() && newData.child('zoneTag').val() === data.child('zoneTag').val())
        )"
      }
    },
    
    // Zone-based posts index - only allow writes where the index zone matches the post's zoneTag
    "posts_by_zone": {
      "$zone": {
        "$postId": {
          ".read": true,
          ".write": "(
             // Allow create/update only when the referenced zone equals the zoneTag on the posted object
             (!data.exists() && newData.hasChild('zoneTag') && newData.child('zoneTag').val() === $zone)
             || (data.exists() && newData.child('zoneTag').val() === data.child('zoneTag').val() && newData.child('zoneTag').val() === $zone)
          )"
        }
      }
    },
    
    // User data - only owners can write
    "users": {
      "$uid": {
        ".read": true,
        ".write": true,
        
        // User's posts index
        "posts": {
          "$postId": {
            ".read": true,
            ".write": true
          }
        },
        
        // User profile
        "profile": {
          ".read": true,
          ".write": true
        },
        
        // User stats
        "stats": {
          ".read": true,
          ".write": true
        },
        
        // Location history
        "locationHistory": {
          ".read": true,
          ".write": true
        }
      }
    },
    
    // Deny all other paths
    ".read": false,
    ".write": false
  }
}
